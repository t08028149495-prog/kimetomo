async function resizeImage(file){
  if(!['image/jpeg','image/png','image/webp'].includes(file.type))throw Error('JPEG・PNG・WebP画像のみ送信できます。動画・GIFは送信できません。');
  if(!file.size||file.size>20*1024*1024)throw Error('元画像は20MB以下にしてください。');
  const url=URL.createObjectURL(file);
  try{
    const img=new Image();img.src=url;await img.decode();
    if(!img.naturalWidth||!img.naturalHeight)throw Error('画像を読み込めませんでした。');
    const scale=Math.min(1,640/img.naturalWidth,480/img.naturalHeight);
    const canvas=document.createElement('canvas');canvas.width=Math.max(1,Math.floor(img.naturalWidth*scale));canvas.height=Math.max(1,Math.floor(img.naturalHeight*scale));
    canvas.getContext('2d').drawImage(img,0,0,canvas.width,canvas.height);
    const blob=await new Promise(resolve=>canvas.toBlob(resolve,'image/jpeg',0.75));
    if(!blob)throw Error('画像の縮小に失敗しました。');
    if(blob.size>2*1024*1024)throw Error('画像が2MBを超えています。');
    return new File([blob],'image.jpg',{type:'image/jpeg'});
  }finally{URL.revokeObjectURL(url)}
}

    } finally {
      btn.disabled = false;
    }
  };

  $('add-room-member-btn').onclick = async () => {
    const room = $('admin-room-select').value;

    const manualId = $('admin-user-id').value.trim();

    const user = manualId ||
      $('admin-user-select').value;

    if (!room || !user) {
      note(
        'ルームと友達を選択してください',
        'error'
      );
      return;
    }

    if (!/^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i.test(user)) {
      note('ユーザーIDの形式が正しくありません', 'error');
      return;
    }

    busy('add-room-member-btn', true);

    try {
      const { error } = await client.rpc(
        'circle_add_member',
        {
          p_room: room,
          p_user: user
        }
      );

      if (error) throw error;

      $('admin-user-id').value = '';

      note(
        'メンバーを追加しました',
        'success'
      );

      await loadRooms();
      await refreshRoomAdmin();

    } catch (e) {
      note(
        'メンバー追加失敗: ' + safeError(e),
        'error'
      );
    } finally {
      busy('add-room-member-btn', false);
    }
  };

  $('refresh-room-admin-btn').onclick =
    refreshRoomAdmin;

  await refreshRoomAdmin();
}

async function refreshRoomAdmin() {
  if (!roomAdminReady || !current) return;

  const roomSelect = $('admin-room-select');
  const userSelect = $('admin-user-select');

  const previousRoom = roomSelect.value;
  const previousUser = userSelect.value;

  roomSelect.replaceChildren();
  userSelect.replaceChildren();

  try {
    const { data: rooms, error: roomError } =
      await client
        .from('circle_rooms')
        .select('id,name')
        .order('name');

    if (roomError) throw roomError;

    for (const room of rooms || []) {
      const option = document.createElement('option');
      option.value = room.id;
      option.textContent = room.name;
      roomSelect.append(option);
    }

    if (previousRoom &&
        [...roomSelect.options].some(
          o => o.value === previousRoom
        )) {
      roomSelect.value = previousRoom;
    }

  } catch (e) {
    note(
      'ルーム一覧取得失敗: ' + safeError(e),
      'error'
    );
  }

  try {
    const { data: users, error: userError } =
      await client
        .from('circle_profiles')
        .select('user_id,display_name,status')
        .eq('status', 'approved')
        .order('display_name');

    if (userError) throw userError;

    for (const user of users || []) {
      const option = document.createElement('option');
      option.value = user.user_id;
      option.textContent =
        user.display_name || user.user_id;
      userSelect.append(option);
    }

    if (previousUser &&
        [...userSelect.options].some(
          o => o.value === previousUser
        )) {
      userSelect.value = previousUser;
    }

  } catch (e) {
    note(
      '友達一覧を取得できません。ユーザーIDの直接入力を利用してください。詳細: ' +
      safeError(e),
      'error'
    );
  }
}

/* キメ友 v1.1 チャット表示拡張 */
(() => {
  'use strict';

  const feed = document.getElementById('chat-feed');

  if (!feed) return;

  const originalDate = date =>
    new Date(date).toLocaleDateString('ja-JP', {
      year: 'numeric',
      month: 'long',
      day: 'numeric'
    });

  function upgradeMessages() {
    const bubbles = Array.from(
      feed.querySelectorAll(':scope > .bubble')
    );

    if (!bubbles.length) return;

    let lastDate = null;

    for (const bubble of bubbles) {
      const meta = bubble.querySelector('.small');

      if (!meta) continue;

      const metaText = meta.textContent || '';
      const isMine = bubble.classList.contains('mine');

      const separator = metaText.indexOf(' · ');
      const sender = separator >= 0
        ? metaText.slice(0, separator)
        : (isMine ? '自分' : 'メンバー');

      const dateText = separator >= 0
        ? metaText.slice(separator + 3)
        : '';

      const parsedDate = new Date(dateText);
      const validDate = !Number.isNaN(parsedDate.getTime());

      const dateLabel = validDate
        ? originalDate(parsedDate)
        : '';

      if (dateLabel && dateLabel !== lastDate) {
        const dateDivider = document.createElement('div');
        dateDivider.className = 'chat-date';
        dateDivider.textContent = dateLabel;
        bubble.before(dateDivider);
        lastDate = dateLabel;
      }

      const message = document.createElement('div');
      message.className =
        'chat-message' + (isMine ? ' mine' : '');

      const avatar = document.createElement('div');
      avatar.className = 'chat-avatar';
      avatar.textContent = sender.charAt(0) || '友';

      const content = document.createElement('div');
      content.className = 'chat-content';

      if (!isMine) {
        const name = document.createElement('div');
        name.className = 'chat-sender';
        name.textContent = sender;
        content.append(name);
      }

      const body = document.createElement('div');
      body.className = 'chat-bubble';

      for (const node of Array.from(bubble.childNodes)) {
        if (node !== meta) {
          body.append(node);
        }
      }

      const time = document.createElement('div');
      time.className = 'chat-time';

      time.textContent = validDate
        ? parsedDate.toLocaleTimeString('ja-JP', {
            hour: '2-digit',
            minute: '2-digit'
          })
        : dateText;

      content.append(body, time);
      message.append(avatar, content);
      bubble.replaceWith(message);
    }

    feed.scrollTop = feed.scrollHeight;
  }

  const observer = new MutationObserver(() => {
    observer.disconnect();
    upgradeMessages();

    observer.observe(feed, {
      childList: true
    });
  });

  observer.observe(feed, {
    childList: true
  });

  upgradeMessages();
})();
</script>
